async function buildReceiptPdf(op:Op){
  const TARGET_BYTES = 24*1024 // 24576 bytes exatos

  const drawPdf = async (imgData:string|null) => {
    const pdf=new jsPDF({orientation:'portrait',unit:'mm',format:'a4'})
    if(imgData){
      try{ pdf.addImage(imgData,'JPEG',160,9,30,15,undefined,'SLOW') }catch{}
    }
    const _d=new Date(op.completedAt||op.date)
    const _p=(n:number)=>String(n).padStart(2,'0')
    const _dateStr=Number.isNaN(_d.getTime())?String(op.completedAt||op.date):`${_d.getFullYear()}.${_p(_d.getDate())}.${_p(_d.getMonth()+1)} ${_p(_d.getHours())}:${_p(_d.getMinutes())}:${_p(_d.getSeconds())} WAT`
    pdf.setFont('helvetica','normal')
    pdf.setFontSize(6.09)
    pdf.setTextColor(0,0,0)
    pdf.text('Digitally signed by noreply@mcxexpress.co.ao',12,14)
    pdf.text(`Date: ${_dateStr}`,12,18)
    pdf.setFontSize(14)
    pdf.setTextColor(200,0,0)
    pdf.text('Comprovativo Digital',105,30,{align:'center'})
    pdf.setDrawColor(225,225,225)
    pdf.setLineWidth(0.35)
    pdf.line(12,38,198,38)
    pdf.setFont('helvetica','normal')
    pdf.setFontSize(11)
    pdf.setTextColor(0,0,0)
    pdf.text('Detalhe da operação realizada através do canal MULTICAIXA Express.',105,46,{align:'center'})
    const rows=[
      ['Data - Hora',formatReceiptDate(op.completedAt||op.date)],
      ['Operação',op.type],
      ['Destinatário',op.holderName||op.recipient],
      ['IBAN',formatIban(op.account)],
      ['Montante',op.amount],
      ['Comissão','-'],
      ['Imposto','-'],
      ['Total',op.amount],
      ['Transacção',op.reference]
    ]
    let y=56
    pdf.setDrawColor(210,0,0)
    pdf.setLineWidth(1.3)
    pdf.line(105,49,105,129)
    rows.forEach(([label,value])=>{
      pdf.setFont('helvetica','bold')
      pdf.setFontSize(11)
      pdf.text(label,101,y,{align:'right'})
      pdf.setFont('helvetica','normal')
      pdf.setFontSize(11.77)
      pdf.text(String(value),111,y)
      y+=8.2
    })
    pdf.setFont('helvetica','normal')
    pdf.setFontSize(11)
    pdf.setTextColor(0,0,0)
    pdf.text('Cuidar do presente, assegurar o futuro.',105,135,{align:'center'})
    pdf.text('BPC - MCX EMV',105,141,{align:'center'})
    pdf.setFillColor(218,218,199)
    pdf.rect(12,189,186,26,'F')
    pdf.setFontSize(8)
    pdf.setTextColor(60,60,60)
    pdf.text('Caso necessite de obter alguma informação, contacte por favor a nossa linha de apoio MULTICAIXA (24h):',105,197,{align:'center'})
    pdf.text('(+244) 222 641 840 | 923 168 840',105,203,{align:'center'})
    pdf.setDrawColor(190,190,175)
    pdf.setLineWidth(0.2)
    pdf.line(20,207,190,207)
    pdf.setFontSize(8)
    pdf.text('IBAN: 0010007100150020007311 | 500290******0477',105,212,{align:'center'})
    return pdf
  }

  // 1. Preparar o logótipo redimensionado uma única vez
  let bitmap:ImageBitmap|null=null
  try{
    const response=await fetch('/assets/bpc-logo-reference.png')
    const blob=await response.blob()
    bitmap=await createImageBitmap(blob)
  }catch{}

  const buildAtQuality = async (quality:number) => {
    let imgData:string|null=null
    if(bitmap){
      const canvas=document.createElement('canvas')
      canvas.width=180;canvas.height=90
      const ctx=canvas.getContext('2d')!
      ctx.fillStyle='#ffffff'
      ctx.fillRect(0,0,180,90)
      ctx.drawImage(bitmap,0,0,180,90)
      imgData=canvas.toDataURL('image/jpeg',quality)
    }
    const pdf=await drawPdf(imgData)
    return pdf.output('blob') as Blob
  }

  // 2. Procurar a qualidade mais alta que ainda fique <= 24KB
  let low=0.2,high=0.95,best:Blob|null=null,bestSize=0
  for(let i=0;i<7;i++){
    const mid=(low+high)/2
    const blob=await buildAtQuality(mid)
    if(blob.size<=TARGET_BYTES){best=blob;bestSize=blob.size;low=mid}
    else{high=mid}
  }
  if(!best){best=await buildAtQuality(0.15);bestSize=best.size}

  // 3. Preencher exatamente até 24KB com um comentário PDF inofensivo (ignorado por qualquer leitor)
  if(bestSize<TARGET_BYTES){
    const bytes=new Uint8Array(await best.arrayBuffer())
    const marker=new TextEncoder().encode('%%EOF')
    let eofPos=-1
    for(let i=bytes.length-marker.length;i>=0;i--){
      let match=true
      for(let j=0;j<marker.length;j++){if(bytes[i+j]!==marker[j]){match=false;break}}
      if(match){eofPos=i;break}
    }
    if(eofPos!==-1){
      const needed=TARGET_BYTES-bestSize
      const padding=new TextEncoder().encode('%'+'0'.repeat(Math.max(0,needed-2))+'\n')
      const finalBytes=new Uint8Array(bestSize+padding.length)
      finalBytes.set(bytes.slice(0,eofPos),0)
      finalBytes.set(padding,eofPos)
      finalBytes.set(bytes.slice(eofPos),eofPos+padding.length)
      return new Blob([finalBytes],{type:'application/pdf'})
    }
  }
  return best
}
