async function buildReceiptPdf(op: Op) {
  const TARGET_BYTES = 24000 // EXATAMENTE 24 KB

  const drawPdf = async (imgData: string | null) => {
    const pdf = new jsPDF({
      orientation: 'portrait',
      unit: 'mm',
      format: 'a4'
    })

    if (imgData) {
      try {
        pdf.addImage(
          imgData,
          'JPEG',
          160,
          9,
          30,
          15,
          undefined,
          'SLOW'
        )
      } catch {}
    }

    const d = new Date(op.completedAt || op.date)
    const p = (n: number) => String(n).padStart(2, '0')

    const dateStr = Number.isNaN(d.getTime())
      ? String(op.completedAt || op.date)
      : `${d.getFullYear()}.${p(d.getDate())}.${p(
          d.getMonth() + 1
        )} ${p(d.getHours())}:${p(d.getMinutes())}:${p(
          d.getSeconds()
        )} WAT`

    pdf.setFont('helvetica', 'normal')
    pdf.setFontSize(6.09)
    pdf.setTextColor(0, 0, 0)

    pdf.text(
      'Digitally signed by noreply@mcxexpress.co.ao',
      12,
      14
    )

    pdf.text(`Date: ${dateStr}`, 12, 18)

    pdf.setFontSize(14)
    pdf.setTextColor(200, 0, 0)

    pdf.text(
      'Comprovativo Digital',
      105,
      30,
      { align: 'center' }
    )

    pdf.setDrawColor(225, 225, 225)
    pdf.setLineWidth(0.35)
    pdf.line(12, 38, 198, 38)

    pdf.setFont('helvetica', 'normal')
    pdf.setFontSize(11)
    pdf.setTextColor(0, 0, 0)

    pdf.text(
      'Detalhe da operação realizada através do canal MULTICAIXA Express.',
      105,
      46,
      { align: 'center' }
    )

    const rows = [
      ['Data - Hora', formatReceiptDate(op.completedAt || op.date)],
      ['Operação', op.type],
      ['Destinatário', op.holderName || op.recipient],
      ['IBAN', formatIban(op.account)],
      ['Montante', op.amount],
      ['Comissão', '-'],
      ['Imposto', '-'],
      ['Total', op.amount],
      ['Transacção', op.reference]
    ]

    let y = 56

    pdf.setDrawColor(210, 0, 0)
    pdf.setLineWidth(1.3)
    pdf.line(105, 49, 105, 129)

    rows.forEach(([label, value]) => {
      pdf.setFont('helvetica', 'bold')
      pdf.setFontSize(11)

      pdf.text(label, 101, y, {
        align: 'right'
      })

      pdf.setFont('helvetica', 'normal')
      pdf.setFontSize(11.77)

      pdf.text(String(value), 111, y)

      y += 8.2
    })

    pdf.setFont('helvetica', 'normal')
    pdf.setFontSize(11)
    pdf.setTextColor(0, 0, 0)

    pdf.text(
      'Cuidar do presente, assegurar o futuro.',
      105,
      135,
      { align: 'center' }
    )

    pdf.text(
      'BPC - MCX EMV',
      105,
      141,
      { align: 'center' }
    )

    pdf.setFillColor(218, 218, 199)
    pdf.rect(12, 189, 186, 26, 'F')

    pdf.setFontSize(8)
    pdf.setTextColor(60, 60, 60)

    pdf.text(
      'Caso necessite de obter alguma informação, contacte por favor a nossa linha de apoio MULTICAIXA (24h):',
      105,
      197,
      { align: 'center' }
    )

    pdf.text(
      '(+244) 222 641 840 | 923 168 840',
      105,
      203,
      { align: 'center' }
    )

    pdf.setDrawColor(190, 190, 175)
    pdf.setLineWidth(0.2)

    pdf.line(20, 207, 190, 207)

    pdf.setFontSize(8)

    pdf.text(
      'IBAN: 0010007100150020007311 | 500290******0477',
      105,
      212,
      { align: 'center' }
    )

    return pdf
  }

  // Carregar logótipo
  let bitmap: ImageBitmap | null = null

  try {
    const response = await fetch(
      '/assets/bpc-logo-reference.png'
    )

    const blob = await response.blob()

    bitmap = await createImageBitmap(blob)
  } catch {}

  // Criar PDF tentando ficar abaixo de 24 KB
  const buildAtQuality = async (quality: number) => {
    let imgData: string | null = null

    if (bitmap) {
      const canvas = document.createElement('canvas')

      canvas.width = 180
      canvas.height = 90

      const ctx = canvas.getContext('2d')!

      ctx.fillStyle = '#ffffff'
      ctx.fillRect(0, 0, 180, 90)

      ctx.drawImage(
        bitmap,
        0,
        0,
        180,
        90
      )

      imgData = canvas.toDataURL(
        'image/jpeg',
        quality
      )
    }

    const pdf = await drawPdf(imgData)

    return pdf.output('blob') as Blob
  }

  // Procurar a maior qualidade possível
  // sem ultrapassar 24.000 bytes
  let low = 0.05
  let high = 0.95

  let best: Blob | null = null

  for (let i = 0; i < 10; i++) {
    const mid = (low + high) / 2

    const blob = await buildAtQuality(mid)

    if (blob.size <= TARGET_BYTES) {
      best = blob
      low = mid
    } else {
      high = mid
    }
  }

  // Fallback
  if (!best) {
    best = await buildAtQuality(0.01)
  }

  // Se já ultrapassou, não é possível simplesmente cortar o PDF.
  // Nesse caso reconstruímos com qualidade mínima.
  if (best.size > TARGET_BYTES) {
    best = await buildAtQuality(0.01)
  }

  // ==========================================================
  // PREENCHER ATÉ EXATAMENTE 24.000 BYTES
  // ==========================================================

  if (best.size < TARGET_BYTES) {
    const bytes = new Uint8Array(
      await best.arrayBuffer()
    )

    const eof = new TextEncoder().encode('%%EOF')

    let eofPos = -1

    // Encontrar o último %%EOF
    for (
      let i = bytes.length - eof.length;
      i >= 0;
      i--
    ) {
      let match = true

      for (let j = 0; j < eof.length; j++) {
        if (bytes[i + j] !== eof[j]) {
          match = false
          break
        }
      }

      if (match) {
        eofPos = i
        break
      }
    }

    if (eofPos !== -1) {
      const needed =
        TARGET_BYTES - best.size

      /*
       * Criamos exatamente "needed" bytes.
       *
       * Cada byte é um "%" (0x25), que representa
       * um comentário válido no formato PDF.
       */

      const padding = new Uint8Array(needed)

      padding.fill(0x25)

      const finalBytes = new Uint8Array(
        TARGET_BYTES
      )

      // Parte antes do %%EOF
      finalBytes.set(
        bytes.slice(0, eofPos),
        0
      )

      // Padding
      finalBytes.set(
        padding,
        eofPos
      )

      // %%EOF e restante
      finalBytes.set(
        bytes.slice(eofPos),
        eofPos + padding.length
      )

      const finalBlob = new Blob(
        [finalBytes],
        {
          type: 'application/pdf'
        }
      )

      // Verificação final
      console.log(
        `PDF final: ${finalBlob.size} bytes`
      )

      console.log(
        `PDF final: ${(finalBlob.size / 1000).toFixed(3)} KB`
      )

      return finalBlob
    }
  }

  // Verificação
  console.log(
    `PDF final: ${best.size} bytes`
  )

  return best
}
